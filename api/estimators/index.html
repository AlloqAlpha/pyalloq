    def __init__(

    base_estimator: BaseReturnEstimator | None = None,

    def estimate_horizon(
    self,
    data: MarketData,
    horizon: int | None = None,
    **kwargs: Any,
) -> pd.DataFrame:
    """
    Returns a 2D DataFrame of shape (H, N) representing forecasted returns
    for stages tau = 0, 1, ..., H - 1.
    """
    h_steps = horizon if horizon is not None else self.horizon
    mu_0 = self.base_estimator.estimate(data, **kwargs)
    assets = list(mu_0.index)
    mu_0_vals = mu_0.to_numpy(dtype=np.float64)

    if self.equilibrium_mode == "grand_mean":
        mu_eq = np.full_like(mu_0_vals, float(np.mean(mu_0_vals)))
    elif self.equilibrium_mode == "historical_mean":
        rets = data.prices.reindex(columns=assets).pct_change().dropna()
        mu_eq = (
            rets.mean().to_numpy(dtype=np.float64) * 252.0
            if len(rets) > 0
            else np.zeros_like(mu_0_vals)
        )
    else:
        mu_eq = np.zeros_like(mu_0_vals)

    rows = []
    persistence = 1.0 - self.alpha_decay
    for tau in range(h_steps):
        decay_factor = persistence**tau
        mu_tau = decay_factor * mu_0_vals + (1.0 - decay_factor) * mu_eq
        rows.append(mu_tau)

    stage_index = pd.Index([f"t+{tau + 1}" for tau in range(h_steps)], name="Stage")
    return pd.DataFrame(rows, index=stage_index, columns=assets)

    VarianceRiskPremiumEstimator ¶

    VarianceRiskPremiumEstimator(
    vrp_aversion: float = 1.5,
    skew_aversion: float = 0.08,
    blend_weight: float = 0.65,
    realized_window: int = 30,
    implied_vol_feature: str = "implied_vol",
    rough_vol_feature: str = "lifted_heston_vol",
    implied_skew_feature: str = "implied_skew",
    annualization_factor: float = 252.0,
)

    Bases: BaseReturnEstimator

    Point-in-Time Variance & Crash-Skewness Risk Premium Return Estimator (VarianceRiskPremiumEstimator, Bollerslev, Tauchen & Zhou 2009; Bakshi, Kapadia & Madan 2003).

    Estimates expected excess asset returns from: 1. Variance Risk Premium (VRP): VRP_{i, t} = sigma_{i, t, IV}^2 - sigma_{i, t, RV}^2 where sigma_{i, t, IV} is the Point-in-Time Model-Free Implied Volatility (data.features["implied_vol"]) and sigma_{i, t, RV} is the Lifted Heston rough realized volatility (data.features["lifted_heston_vol"]) or trailing realized volatility. 2. Crash-Skewness Risk Premium: SkewPrem_{i, t} = -MFIS_{i, t} * sigma_{i, t, IV} where negative risk-neutral skewness (MFIS_{i, t} < 0 from data.features["implied_skew"]) reflects expensive out-of-the-money put protection and commands a positive expected return compensation in equilibrium: E_t[r_{i, t+1}] = r_f + gamma_vrp * VRP_{i, t} + eta_skew * (-MFIS_{i, t}) * sigma_{i, t, IV} + (1 - blend_weight) * mu_{hist, i}

def build_precision_graph_laplacian(

    """
    Estimates the Ledoit-Wolf shrunk correlation matrix, precision matrix,
    non-negative partial correlation adjacency W, and normalized Graph Laplacian L.
    """
    n_obs, n_assets = returns.shape
    if n_assets == 1:
        return (
            np.zeros((1, 1), dtype=np.float64),
            np.zeros((1, 1), dtype=np.float64),
            np.ones((1, 1), dtype=np.float64),
        )

        try:
            lw = LedoitWolf().fit(z_ret)
            corr = lw.covariance_.astype(np.float64)
        except Exception:
            corr = np.cov(z_ret, rowvar=False)
    else:
        corr = np.eye(n_assets, dtype=np.float64)

    d_inv = 1.0 / np.sqrt(np.clip(np.diag(corr), 1e-10, None))
    corr = corr * np.outer(d_inv, d_inv)
    corr = 0.5 * (corr + corr.T)
    np.fill_diagonal(corr, 1.0)

    theta = np.linalg.inv(corr + 1e-3 * np.eye(n_assets, dtype=np.float64))
    theta_diag = np.sqrt(np.clip(np.diag(theta), 1e-10, None))
    partial_corr = -theta / np.outer(theta_diag, theta_diag)

    adjacency = np.maximum(0.0, partial_corr)
    np.fill_diagonal(adjacency, 0.0)

    deg = np.sum(adjacency, axis=1)
    deg_inv_sqrt = np.where(deg > 1e-10, 1.0 / np.sqrt(np.maximum(deg, 1e-10)), 0.0)
    norm_adj = adjacency * np.outer(deg_inv_sqrt, deg_inv_sqrt)
    laplacian = np.eye(n_assets, dtype=np.float64) - norm_adj
    laplacian = 0.5 * (laplacian + laplacian.T)

    return laplacian, norm_adj, corr

    def fit_from_returns(
    self,
    returns: np.ndarray,
) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Extracts jump point-process increments dN_t in {0, 1}^{T x N} from asset returns,
    solves the Nyström Wiener-Hopf system for the excitation kernels Phi(tau), and returns
    `(branching_matrix, spectral_radius, baseline_intensity)`.
    """
    n_obs, n_assets = returns.shape
    if n_obs < self.max_lag + 3:
        zeros = np.zeros((n_assets, n_assets), dtype=np.float64)
        return zeros, 0.0, np.full(n_assets, 0.05, dtype=np.float64)

    std = np.std(returns, axis=0, ddof=1)
    std = np.where(std > 1e-10, std, 1.0)
    z_scores = np.abs(returns - np.mean(returns, axis=0)) / std

    # Discrete jump indicator process dN_{t, i} in {0, 1}
    dN = (z_scores >= self.jump_z_threshold).astype(np.float64)
    mean_rate = np.clip(np.mean(dN, axis=0), 1e-4, 0.95)
    centered_dN = dN - mean_rate[None, :]

    Q = min(self.max_lag, max(1, n_obs // 4))
    # Empirical lag covariance matrices C(h) = E[(dN_{t} - Lambda)(dN_{t-h} - Lambda)^T]
    # C_lags[h - 1] has shape (N, N) for h = 1, ..., Q, plus C_0 at lag 0
    C_0 = (centered_dN.T @ centered_dN) / float(n_obs) + self.ridge_reg * np.eye(
        n_assets, dtype=np.float64
    )
    C_lags = np.zeros((Q, n_assets, n_assets), dtype=np.float64)
    for h in range(1, Q + 1):
        C_lags[h - 1] = (centered_dN[h:].T @ centered_dN[:-h]) / float(n_obs - h)

    # Assemble Nyström block-Toeplitz Wiener-Hopf operator K_WH of shape (Q * N, Q * N)
    # where block (q, r) is C(|q - r|)^T
    block_size = Q * n_assets
    K_wh = np.zeros((block_size, block_size), dtype=np.float64)
    rhs = np.zeros((block_size, n_assets), dtype=np.float64)

    for q in range(Q):
        q_slice = slice(q * n_assets, (q + 1) * n_assets)
        rhs[q_slice, :] = C_lags[q].T
        for r in range(Q):
            r_slice = slice(r * n_assets, (r + 1) * n_assets)
            diff = q - r
            if diff == 0:
                block = C_0
            elif diff > 0:
                block = C_lags[diff - 1]
            else:
                block = C_lags[-diff - 1].T
            K_wh[q_slice, r_slice] = block

    K_wh += self.ridge_reg * np.eye(block_size, dtype=np.float64)
    # Solve K_wh @ Phi_stack = rhs -> Phi_stack[q] = Phi(tau_q)^T
    try:
        phi_stack = np.linalg.solve(K_wh, rhs)
    except np.linalg.LinAlgError:
        phi_stack = np.linalg.lstsq(K_wh, rhs, rcond=None)[0]

    # Project onto non-negative causal excitation kernels and integrate L1 norm over lags
    kernels = np.zeros((Q, n_assets, n_assets), dtype=np.float64)
    for q in range(Q):
        q_slice = slice(q * n_assets, (q + 1) * n_assets)
        kernels[q] = np.maximum(0.0, phi_stack[q_slice, :].T)

    # Quadrature integration G_{ij} = sum_{q=1}^Q w_q phi_{ij}(q)
    # Normalize lag weights so G represents expected offspring per jump event
    branching = np.sum(kernels, axis=0) / float(max(1, Q))

    eigvals = np.linalg.eigvals(branching)
    raw_radius = float(np.max(np.abs(eigvals))) if len(eigvals) > 0 else 0.0
    if raw_radius > self.max_spectral_radius and raw_radius > 1e-12:
        scale = self.max_spectral_radius / raw_radius
        branching = branching * scale
        spec_radius = self.max_spectral_radius
    else:
        spec_radius = raw_radius

    baseline_mu = mean_rate * max(0.05, 1.0 - spec_radius)
    return branching, float(spec_radius), baseline_mu

    self,
    x: torch.Tensor,
    return_jump_intensity: bool = False,

    Parameters
    ----------
    x : torch.Tensor
        Input tensor of shape `(B, L, N, F)` or `(B_flat, L, F)`.

    Returns
    -------

        plus optional terminal Hawkes jump intensity of shape `(B, N)`.

        B, L, N, F = x.shape
        x_flat = x.permute(0, 2, 1, 3).reshape(B * N, L, F)

        x_flat = x

    lam = torch.full((x_flat.shape[0], 1), 0.05, device=x.device, dtype=x.dtype)
    decay = torch.sigmoid(self.hawkes_decay)

        lsig_k = logsig_tokens[:, k, :]
        shock_k = shocks[:, k, :]

        # 1. Continuous Log-ODE step
        dh = self.log_ode_field(torch.cat([h, lsig_k], dim=-1))
        h_cont = h + dh

        exc = torch.nn.functional.softplus(
            self.hawkes_excitation(torch.cat([h_cont, torch.abs(shock_k)], dim=-1))
        )
        lam = decay * lam + (1.0 - decay) * exc

        if return_jump_intensity:
            return h_out, lam.view(B, N)

    if return_jump_intensity:

    def forward(
    self,
    node_embeddings: torch.Tensor,
    buss_vilkov_target: torch.Tensor | None = None,
    fiedler_value: torch.Tensor | float = 0.2,
    hawkes_radius: torch.Tensor | float = 0.2,
) -> torch.Tensor:
    """
    Parameters
    ----------
    node_embeddings : torch.Tensor
        Spatially attended asset embeddings of shape `(B, N, d)`.
    buss_vilkov_target : torch.Tensor, optional
        Optional target covariance matrix of shape `(B, N, N)` or `(N, N)`.
        When None, constructs a variance-matched Buss-Vilkov equicorrelation target.
    """

    device = node_embeddings.device
    dtype = node_embeddings.dtype

    q = self.proj_q(node_embeddings)
    k = self.proj_k(node_embeddings)
    raw_mat = torch.bmm(q, k.transpose(1, 2)) / (self.hidden_dim**0.5)

    L_strict = torch.tril(raw_mat, diagonal=-1)
    d_raw = torch.diagonal(raw_mat, dim1=-2, dim2=-1)
    d_pos = torch.nn.functional.softplus(d_raw) + self.eps
    L = L_strict + torch.diag_embed(d_pos)

    eye = torch.eye(N, device=device, dtype=dtype).unsqueeze(0)
    sigma_raw = torch.bmm(L, L.transpose(1, 2)) + self.eps * eye

    if buss_vilkov_target is not None:
        if buss_vilkov_target.dim() == 2:
            sigma_target = buss_vilkov_target.unsqueeze(0).expand(B, N, N)
        else:
            sigma_target = buss_vilkov_target
    else:
        # Construct PSD equicorrelation Buss-Vilkov prior from predicted marginal volatilities
        diag_var = torch.diagonal(sigma_raw, dim1=-2, dim2=-1).clamp(min=self.eps)
        std_vec = torch.sqrt(diag_var)
        outer_std = std_vec.unsqueeze(-1) * std_vec.unsqueeze(-2)
        rho = self.default_equicorr
        equi_corr = (1.0 - rho) * eye + rho * torch.ones(
            (1, N, N), device=device, dtype=dtype
        )
        sigma_target = outer_std * equi_corr

    alpha = self.compute_shrinkage_alpha(
        fiedler_value=fiedler_value,
        hawkes_radius=hawkes_radius,
        device=device,
        dtype=dtype,
    )
    while alpha.dim() < 3:
        alpha = alpha.unsqueeze(-1)

    sigma_shrunk = (1.0 - alpha) * sigma_raw + alpha * sigma_target
    return 0.5 * (sigma_shrunk + sigma_shrunk.transpose(1, 2))

    def compute_von_neumann_entropy(
    cov: torch.Tensor,
    normalize: bool = True,
) -> torch.Tensor:
    """
    Computes the Von Neumann Entropy (CVNE) of a batch of SPD covariance matrices
    (PDF p. 8, Kronecker coVariance Neural Network regularity metric):
        S_vN(Sigma) = - sum_{i=1}^N p_i log(p_i),   where p_i = lambda_i / Tr(Sigma).

    When `normalize=True`, divides by `log(N)` so `S_vN in [0, 1]`:
    - `S_vN -> 1`: isotropic / well-diversified eigenvalue spectrum.
    - `S_vN -> 0`: rank-1 contagion collapse (single dominant market mode).
    """
    squeezed = cov.dim() == 2
    if squeezed:
        cov = cov.unsqueeze(0)

    B, N, _ = cov.shape
    if N <= 1:
        out = torch.ones(B, device=cov.device, dtype=cov.dtype)
        return out.squeeze(0) if squeezed else out

    sym_cov = 0.5 * (cov + cov.transpose(-1, -2))
    eigvals = torch.linalg.eigvalsh(sym_cov).clamp(min=1e-8)
    probs = eigvals / eigvals.sum(dim=-1, keepdim=True).clamp(min=1e-8)
    entropy = -(probs * torch.log(probs.clamp(min=1e-12))).sum(dim=-1)
    if normalize:
        entropy = entropy / float(np.log(max(2, N)))
    return entropy.squeeze(0) if squeezed else entropy
