    def efficient_frontier(
    self,
    data: MarketData,
    cov_matrix: pd.DataFrame,
    expected_returns: pd.Series,
    n_points: int = 20,
) -> list[OptimizationResult]:
    """
    Traces out `n_points` portfolios along the Mean-Variance Efficient Frontier
    between the Minimum Volatility portfolio return and the maximum feasible return.
    """
    if n_points < 2:
        raise ValueError("n_points must be >= 2.")

    assets = list(cov_matrix.columns)
    mu_vals = np.asarray(expected_returns.reindex(assets).values, dtype=float)

    min_vol_alloc = MarkowitzAllocator(
        tickers=assets,
        objective=ObjectiveFunction.MIN_VOLATILITY,
        weight_bounds=self.weight_bounds,
        market_neutral=self.market_neutral,
        max_leverage=self.max_leverage,
        l2_gamma=self.l2_gamma,
        sector_mapper=self.sector_mapper,
        sector_lower=self.sector_lower,
        sector_upper=self.sector_upper,
    )
    res_min = min_vol_alloc.allocate(
        data, cov_matrix=cov_matrix, expected_returns=expected_returns
    )
    min_ret = (
        float(res_min.expected_return)
        if res_min.expected_return is not None
        else float(np.min(mu_vals))
    )

    lb, ub = self._resolve_bounds(assets)
    w_max = cp.Variable(len(assets))
    c_max = self._build_standard_constraints(w_max, assets, data)
    prob_max = cp.Problem(cp.Maximize(mu_vals @ w_max), c_max)
    try:
        prob_max.solve()
    except cp.SolverError:
        prob_max.solve(solver=cp.SCS)
    max_ret = (
        float(prob_max.value)
        if prob_max.status in ("optimal", "optimal_inaccurate")
        and prob_max.value is not None
        else float(np.max(mu_vals))
    )

    target_grid = np.linspace(min_ret, max_ret - 1e-6, n_points)
    frontier: list[OptimizationResult] = []
    for tr in target_grid:
        ef_alloc = MarkowitzAllocator(
            tickers=assets,
            objective=ObjectiveFunction.EFFICIENT_RETURN,
            weight_bounds=self.weight_bounds,
            market_neutral=self.market_neutral,
            max_leverage=self.max_leverage,
            target_return=float(tr),
            l2_gamma=self.l2_gamma,
            sector_mapper=self.sector_mapper,
            sector_lower=self.sector_lower,
            sector_upper=self.sector_upper,
        )
        res = ef_alloc.allocate(
            data, cov_matrix=cov_matrix, expected_returns=expected_returns
        )
        if "OPTIMAL" in res.status:
            frontier.append(res)
    return frontier

    MeanCVaRAllocator ¶

    MeanCVaRAllocator(
    tickers: list[str] | None = None,
    beta: float = 0.95,
    objective: ObjectiveFunction = ObjectiveFunction.MIN_CVAR,
    weight_bounds: tuple[float, float]
    | dict[str, tuple[float, float]] = (0.0, 1.0),
    market_neutral: bool = False,
    max_leverage: float | None = None,
    target_return: float | None = None,
    risk_aversion: float | None = None,
    l2_gamma: float = 0.0,
    max_turnover: float | None = None,
)

    Bases: BaseAllocator

    Single-period Conditional Value-at-Risk (CVaR / Expected Shortfall) Portfolio Optimizer using the Rockafellar & Uryasev (2000) convex linear programming formulation:

    CVaR_alpha(w) = alpha_var + (1 / ((1 - beta) * T)) * sum_{t=1}^T u_t
subject to: u_t >= -r_t^T w - alpha_var, u_t >= 0.

    Supports: - ObjectiveFunction.MIN_CVAR: Minimize portfolio CVaR (optionally subject to target_return). - ObjectiveFunction.EFFICIENT_RETURN: Minimize portfolio CVaR subject to mu^T w >= target_return. - ObjectiveFunction.MAX_RETURN: Maximize mu^T w - risk_aversion * CVaR_beta(w). - Custom weight_bounds, market_neutral, max_leverage, max_turnover, and l2_gamma.

    regime_gate: Any | None = None,
)

    Bases: BaseAllocator

    Model Predictive Control (MPC) Multi-Period Portfolio Optimizer (Boyd et al., 2017; Gârleanu & Pedersen, 2013; Rosenberg et al., 2016).

    Solves for a finite-horizon sequence of post-trade weights X = [x_{t+1}, ..., x_{t+H}] and trade vectors Z = [z_t, ..., z_{t+H-1}] with linearized dynamics x_{tau+1} = x_tau + z_tau:

    Maximize sum_{tau=t}^{t+H-1} beta^{tau-t} [
    mu_tau^T x_{tau+1}
    - gamma * x_{tau+1}^T Sigma_tau x_{tau+1}
    - phi_hold(x_{tau+1})
    - phi_trade(z_tau)
    - permanent_impact(x_tau, z_tau)
]

    Supports: - Proportional Transaction Costs (p = 1): L1 bid-ask & commission (no-trade region). - Market Impact Power Cone (p = 3/2): |z_{tau, i}|^{3/2} child-order splitting. - Quadratic Temporary & Permanent Impact (p = 2, Gârleanu-Pedersen state-space). - Dual Solver Backends: * backend="cvxpy": Conic interior-point formulation (Rotated Quadratic & Power Cones). * backend="admm": Alternating Direction Method of Multipliers with closed-form Soft-Thresholding (Section 3.1) for fast L1-constrained trajectory optimization.

    def forward(
    self,
    mu_seq: torch.Tensor,
    cov_seq: torch.Tensor | None = None,
    w_current: torch.Tensor | None = None,
) -> torch.Tensor:
    """
    Parameters
    ----------
    mu_seq : torch.Tensor
        Predicted multi-horizon expected returns of shape (B, H, N) or (B, N).
    cov_seq : torch.Tensor, optional
        Covariance trajectory of shape (B, H, N, N) or (B, N, N). Defaults to identity * 0.01.
    w_current : torch.Tensor, optional
        Pre-trade starting weights of shape (B, N). Defaults to equal weights 1/N.

    Returns
    -------
    torch.Tensor
        Optimal multi-period weight trajectory z* of shape (B, H, N).
    """
    if mu_seq.dim() == 2:
        mu_seq = mu_seq.unsqueeze(1).expand(-1, self.horizon, -1)

    B, H, N = mu_seq.shape
    device = mu_seq.device
    dtype = mu_seq.dtype

    if cov_seq is None:
        eye = torch.eye(N, device=device, dtype=dtype) * 0.01
        cov_seq = eye.view(1, 1, N, N).expand(B, H, N, N)
    elif cov_seq.dim() == 3:
        cov_seq = cov_seq.unsqueeze(1).expand(B, H, N, N)

    if w_current is None:
        w_current = torch.full((B, N), 1.0 / N, device=device, dtype=dtype)

    if self.implicit_backward:
        return _ImplicitMDFPFunction.apply(
            mu_seq,
            cov_seq,
            w_current,
            self.risk_aversion,
            self.cost_penalty,
            self.step_size,
            self.max_iter,
            self.tol,
            self.discount,
            self.smooth_eps,
            self.neumann_steps,
            self.neumann_damping,
        )

    # Unrolled fallback if implicit_backward=False
    w_init = w_current.clamp(min=1e-6)
    w_init = w_init / w_init.sum(dim=-1, keepdim=True)
    z = w_init.unsqueeze(1).expand(B, H, N).clone()
    for _ in range(self.max_iter):
        z = _mdfp_step(
            z=z,
            mu_seq=mu_seq,
            cov_seq=cov_seq,
            w_current=w_current,
            gamma=self.risk_aversion,
            kappa=self.cost_penalty,
            eta=self.step_size,
            discount=self.discount,
            smooth_eps=self.smooth_eps,
        )
    return z

    """
    Evaluates the policy directly on a Gymnasium observation from `POMDPPortfolioEnv`,
    `PortfolioGymEnv`, or `ScenarioPortfolioEnv`.
    """
    self.eval()
    device = next(self.parameters()).device
    dtype = next(self.parameters()).dtype

    if isinstance(observation, dict):
        raw_mkt = observation.get(
            "market_state",
            observation.get("market", observation.get("returns")),
        )
        mkt = np.asarray(raw_mkt)
        w_np = observation.get("current_weights", observation.get("weights"))
    else:
        mkt = np.asarray(observation)
        w_np = None

    if mkt.ndim == 2:
        x_t = (
            torch.tensor(mkt, device=device, dtype=dtype).unsqueeze(0).unsqueeze(-1)
        )
    elif mkt.ndim == 3:
        x_t = torch.tensor(mkt, device=device, dtype=dtype).unsqueeze(0)
    elif mkt.ndim == 4:
        x_t = torch.tensor(mkt, device=device, dtype=dtype)
    else:
        raise ValueError(f"Unsupported market observation shape: {mkt.shape}")

    w_t: torch.Tensor | None = None
    if w_np is not None:
        w_arr = np.asarray(w_np)
        if w_arr.ndim == 1:
            w_t = torch.tensor(w_arr, device=device, dtype=dtype).unsqueeze(0)
        else:
            w_t = torch.tensor(w_arr, device=device, dtype=dtype)

    w_out = self.forward(x_t, w_current=w_t)
    action = w_out.squeeze(0).cpu().numpy()
    return action, {"deterministic": deterministic}

    Rough-Path Signature Maximum Mean Discrepancy (Signature-MMD) Regime Gate.

    Unlike Gaussian Bures-Wasserstein distance (which only compares first two point-in-time moments), Signature-MMD compares the expected Lyndon Depth-2 Log-Signature of rolling sub-path trajectories between a fast (recent) window and a slow (structural) window: MMD_sig^2(P_fast, P_slow) = || E_{X ~ P_fast}[LogSig(X)] - E_{Y ~ P_slow}[LogSig(Y)] ||_2^2

    Because Depth-2 Log-Signatures include the antisymmetric Levy area A^{i,j} = 0.5 * sum_t (X_{t-1}^i dX_t^j - X_{t-1}^j dX_t^i), Signature-MMD detects lead-lag directionality inversions and rough path-geometry shifts even when marginal means and covariances remain unchanged.

    @staticmethod
def compute_subpath_log_signatures(
    returns_arr: np.ndarray,
    subpath_window: int = 8,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Extracts sliding-window depth-2 Lyndon log-signatures from a 2D return array (T, N).

    Returns:
        full_logsigs: shape (M, d + d(d-1)/2) where d = N + 1 (time-augmented)
        levy_areas: shape (M, d(d-1)/2)
    """
    t_len, n_assets = returns_arr.shape
    w = min(max(2, subpath_window), t_len)
    if t_len < w:
        d = n_assets + 1
        n_pairs = (d * (d - 1)) // 2
        return np.zeros((1, d + n_pairs), dtype=np.float64), np.zeros(
            (1, n_pairs), dtype=np.float64
        )

    vol_scale = np.maximum(np.std(returns_arr, axis=0, keepdims=True), 1e-6)
    norm_ret = np.clip(returns_arr / vol_scale, -10.0, 10.0)
    d = n_assets + 1
    pair_i, pair_j = np.triu_indices(d, k=1)
    n_sub = t_len - w + 1
    full_list: list[np.ndarray] = []
    levy_list: list[np.ndarray] = []
    dt_col = np.full((w, 1), 1.0 / float(w), dtype=np.float64)

    for start in range(n_sub):
        seg = norm_ret[start : start + w]
        dx = np.hstack([dt_col, seg])  # (w, d)
        x_cum = np.cumsum(dx, axis=0)
        x_prev = np.vstack([np.zeros((1, d), dtype=np.float64), x_cum[:-1]])
        s1 = x_cum[-1]
        levy_mat = 0.5 * (
            np.einsum("ti,tj->ij", x_prev, dx) - np.einsum("tj,ti->ij", x_prev, dx)
        )
        levy_vec = levy_mat[pair_i, pair_j]
        full_list.append(np.concatenate([s1, levy_vec]))
        levy_list.append(levy_vec)

    return np.vstack(full_list), np.vstack(levy_list)
