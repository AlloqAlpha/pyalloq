    def split(
    self, data: MarketData
) -> Generator[tuple[MarketData, MarketData], None, None]:
    """Yields (train_data, test_data) for neural network training."""

    total_length = len(data.prices)
    fold_size = total_length // self.n_splits
    embargo_size = int(total_length * self.embargo_pct)

    indices = np.arange(total_length)
    dates = data.prices.index

    for i in range(self.n_splits):
        test_start = i * fold_size
        test_end = test_start + fold_size if i < self.n_splits - 1 else total_length

        test_indices = indices[test_start:test_end]
        train_indices_left = indices[: max(0, test_start - embargo_size)]
        train_indices_right = indices[min(total_length, test_end + embargo_size) :]

        train_indices = np.concatenate([train_indices_left, train_indices_right])

        if len(train_indices) == 0:
            continue

        train_dates = dates[train_indices]
        test_dates = dates[test_indices]

        train_data = MarketData(
            prices=data.prices.loc[train_dates],
            features={k: v.loc[train_dates] for k, v in data.features.items()}
            if data.features
            else {},
            cross_sectional=data.cross_sectional,
            risk_free_rate=data.risk_free_rate.loc[train_dates]
            if isinstance(data.risk_free_rate, pd.Series)
            else data.risk_free_rate,
        )

        test_data = MarketData(
            prices=data.prices.loc[test_dates],
            features={k: v.loc[test_dates] for k, v in data.features.items()}
            if data.features
            else {},
            cross_sectional=data.cross_sectional,
            risk_free_rate=data.risk_free_rate.loc[test_dates]
            if isinstance(data.risk_free_rate, pd.Series)
            else data.risk_free_rate,
        )

        yield train_data, test_data

    def decompose_costs(
    self,
    weights_delta: CostData,
    features_slice: dict[str, CostData] | None = None,
) -> tuple[CostData, CostData]:
    """
    Decomposes transaction costs into (proportional_spread_cost, nonlinear_impact_cost).
    Defaults to attributing 100% of `calculate_costs` to proportional spread/commission.
    """
    spread_cost = self.calculate_costs(weights_delta, features_slice)
    impact_cost = spread_cost * 0.0  # type: ignore[operator]
    return spread_cost, impact_cost

    def split(
    self, data: MarketData
) -> list[tuple[MarketData, MarketData, tuple[int, ...]]]:
    """
    Generates all C(N, k) purged & embargoed train/test splits of MarketData.
    """
    total_length = len(data.prices)
    indices = np.arange(total_length)
    groups = np.array_split(indices, self.n_groups)
    purge_size = int(total_length * self.purge_pct)
    embargo_size = int(total_length * self.embargo_pct)
    dates = data.prices.index

    splits_out: list[tuple[MarketData, MarketData, tuple[int, ...]]] = []

    for test_group_ids in combinations(range(self.n_groups), self.k_test_groups):
        test_mask = np.zeros(total_length, dtype=bool)
        banned_mask = np.zeros(total_length, dtype=bool)

        for gid in test_group_ids:
            g_idx = groups[gid]
            if len(g_idx) == 0:
                continue
            test_mask[g_idx] = True
            start_i = max(0, int(g_idx[0]) - purge_size)
            end_i = min(total_length, int(g_idx[-1]) + 1 + embargo_size)
            banned_mask[start_i:end_i] = True

        train_mask = ~banned_mask
        train_dates = dates[train_mask]
        test_dates = dates[test_mask]

        if len(train_dates) == 0 or len(test_dates) == 0:
            continue

        train_data = MarketData(
            prices=data.prices.loc[train_dates],
            features={k: v.loc[train_dates] for k, v in data.features.items()}
            if data.features
            else {},
            cross_sectional=data.cross_sectional,
            risk_free_rate=data.risk_free_rate.loc[train_dates]
            if isinstance(data.risk_free_rate, pd.Series)
            else data.risk_free_rate,
        )
        test_data = MarketData(
            prices=data.prices.loc[test_dates],
            features={k: v.loc[test_dates] for k, v in data.features.items()}
            if data.features
            else {},
            cross_sectional=data.cross_sectional,
            risk_free_rate=data.risk_free_rate.loc[test_dates]
            if isinstance(data.risk_free_rate, pd.Series)
            else data.risk_free_rate,
        )
        splits_out.append((train_data, test_data, test_group_ids))

    return splits_out

    def assemble_cpcv_paths(
    self,
    oos_split_returns: dict[tuple[int, ...], pd.Series],
    historical_returns: pd.Series | None = None,
    risk_free_rate: float = 0.0,
) -> CPCVResult:
    """
    Reconstructs the phi(N, k) = C(N - 1, k - 1) full-horizon out-of-sample backtest paths
    from the C(N, k) test split evaluations according to Lopez de Prado (2018).
    """
    group_appearances: dict[int, list[pd.Series]] = {
        g: [] for g in range(self.n_groups)
    }

    # Sort splits lexicographically so path assignment is deterministic
    for test_group_ids in sorted(oos_split_returns.keys()):
        split_ret = oos_split_returns[test_group_ids]
        # Split the test return series evenly across its k test groups
        group_slices = np.array_split(
            np.arange(len(split_ret)), len(test_group_ids)
        )
        for gid, idx_slice in zip(test_group_ids, group_slices):
            group_appearances[gid].append(split_ret.iloc[idx_slice])

    n_paths = min(len(v) for v in group_appearances.values() if len(v) > 0)
    path_series_list: list[pd.Series] = []
    block_orders: list[tuple[int, ...]] = []

    for p_idx in range(n_paths):
        segments = [group_appearances[g][p_idx] for g in range(self.n_groups)]
        combined = pd.concat(segments).sort_index()
        combined.name = f"CPCV_Path_{p_idx}"
        path_series_list.append(combined)
        block_orders.append(tuple(range(self.n_groups)))

    returns_df = pd.concat(path_series_list, axis=1).dropna(how="all").fillna(0.0)
    return self._build_cpcv_result(
        returns_df,
        block_orders=block_orders,
        historical_returns=historical_returns,
        risk_free_rate=risk_free_rate,
    )

    def scramble_timelines(
    self,
    source: BacktestResult | pd.Series,
    risk_free_rate: float = 0.0,
    block_bootstrap: bool = True,
) -> CPCVResult:
    """
    Partitions historical strategy returns into `n_groups` regime blocks and generates
    combinatorial scrambled timelines (permutations + block-resampled paths with subtle
    within-block jitter/subsampling) to evaluate strategy survival across scrambled
    chronological sequences of market regimes.
    """
    if isinstance(source, BacktestResult):
        hist_returns = source.returns.dropna()
    else:
        hist_returns = pd.Series(source).dropna()

    rng = np.random.default_rng(self.random_state)
    ret_vals = hist_returns.to_numpy(dtype=np.float64)
    blocks = np.array_split(ret_vals, self.n_groups)
    n_total = len(ret_vals)

    # Generate distinct block sequences (combinatorial permutations + bootstrap block draws)
    all_orders: list[tuple[int, ...]] = []
    if self.n_groups <= 6:
        from itertools import permutations

        perms = list(permutations(range(self.n_groups)))
        # Exclude identity as path 0 if we want pure scrambled, or include all
        if len(perms) > self.max_paths:
            chosen_idx = rng.choice(len(perms), size=self.max_paths, replace=False)
            all_orders = [perms[i] for i in chosen_idx]
        else:
            all_orders = perms
    else:
        seen: set[tuple[int, ...]] = set()
        for _ in range(self.max_paths * 3):
            if len(all_orders) >= self.max_paths:
                break
            if block_bootstrap:
                order = tuple(
                    int(x)
                    for x in rng.choice(
                        self.n_groups, size=self.n_groups, replace=True
                    )
                )
            else:
                order = tuple(int(x) for x in rng.permutation(self.n_groups))
            if order not in seen:
                seen.add(order)
                all_orders.append(order)

    scrambled_cols: dict[str, np.ndarray] = {}
    for idx, order in enumerate(all_orders):
        recombined = np.concatenate([blocks[b] for b in order])
        if block_bootstrap and idx > 0:
            # Apply circular shift within each block so paths aren't identical in terminal return
            shifted_blocks = []
            for b in order:
                blk = blocks[b]
                if len(blk) > 2:
                    shift = int(rng.integers(0, len(blk)))
                    # Sample with replacement 90% + permutation to vary compounding & Sharpe slightly
                    sub_idx = rng.choice(len(blk), size=len(blk), replace=True)
                    blk = np.roll(blk, shift) if idx % 2 == 0 else blk[sub_idx]
                shifted_blocks.append(blk)
            recombined = np.concatenate(shifted_blocks)

        # Align length to n_total
        if len(recombined) >= n_total:
            recombined = recombined[:n_total]
        else:
            pad = np.zeros(n_total - len(recombined))
            recombined = np.concatenate([recombined, pad])

        scrambled_cols[f"Path_{idx}"] = recombined

    returns_df = pd.DataFrame(scrambled_cols, index=hist_returns.index)
    return self._build_cpcv_result(
        returns_df,
        block_orders=all_orders,
        historical_returns=hist_returns,
        risk_free_rate=risk_free_rate,
    )

    def evaluate_cscv(
    self,
    strategy_returns: pd.DataFrame | np.ndarray | Sequence[BacktestResult],
) -> PBOResult:
    """
    Calculates exact PBO via Combinatorial Symmetric Cross-Validation (CSCV) across M trials.
    """
    if isinstance(strategy_returns, pd.DataFrame):
        M_mat = (
            strategy_returns.dropna(how="all")
            .fillna(0.0)
            .to_numpy(dtype=np.float64)
        )
    elif isinstance(strategy_returns, np.ndarray):
        M_mat = np.asarray(strategy_returns, dtype=np.float64)
    else:
        df_concat = pd.concat(
            [
                res.returns.rename(f"{res.name}_{i}")
                for i, res in enumerate(strategy_returns)
            ],
            axis=1,
        ).fillna(0.0)
        M_mat = df_concat.to_numpy(dtype=np.float64)

    if M_mat.ndim != 2 or M_mat.shape[1] < 2:
        raise ValueError(
            "CSCV requires a 2D return matrix of shape (T, M) with at least M >= 2 strategy trials."
        )

    T, M_trials = M_mat.shape
    S = min(self.n_partitions, (T // 5) * 2)
    S = max(2, S - (S % 2))

    submatrices = np.array_split(M_mat, S, axis=0)
    half_S = S // 2
    all_indices = set(range(S))

    logits_list: list[float] = []
    is_sharpes_list: list[float] = []
    oos_sharpes_list: list[float] = []

    for is_group_tuple in combinations(range(S), half_S):
        oos_group_tuple = tuple(sorted(all_indices - set(is_group_tuple)))
        J_is = np.vstack([submatrices[i] for i in is_group_tuple])
        J_oos = np.vstack([submatrices[i] for i in oos_group_tuple])

        R_is = self._annualized_sharpes(J_is, self.risk_free_rate)
        R_oos = self._annualized_sharpes(J_oos, self.risk_free_rate)

        best_m = int(np.argmax(R_is))
        is_best_sr = float(R_is[best_m])
        oos_of_best_sr = float(R_oos[best_m])

        # Relative rank of the IS-optimal strategy in OOS (in (0, 1))
        less_count = float(np.sum(R_oos < oos_of_best_sr))
        equal_count = float(np.sum(R_oos == oos_of_best_sr))
        omega_c = (less_count + 0.5 * equal_count) / float(M_trials)
        omega_c = float(
            np.clip(omega_c, 1.0 / (2.0 * M_trials), 1.0 - 1.0 / (2.0 * M_trials))
        )

        logit = math.log(omega_c / (1.0 - omega_c))
        logits_list.append(logit)
        is_sharpes_list.append(is_best_sr)
        oos_sharpes_list.append(oos_of_best_sr)

    logits_arr = np.asarray(logits_list, dtype=np.float64)
    is_arr = np.asarray(is_sharpes_list, dtype=np.float64)
    oos_arr = np.asarray(oos_sharpes_list, dtype=np.float64)

    pbo = float(np.mean(logits_arr <= 0.0))
    prob_loss = float(np.mean(oos_arr < 0.0))

    if len(is_arr) > 1 and np.std(is_arr) > 1e-12:
        slope, intercept = np.polyfit(is_arr, oos_arr, 1)
    else:
        slope, intercept = 0.0, float(np.mean(oos_arr))

    full_sharpes = self._annualized_sharpes(M_mat, self.risk_free_rate)
    best_overall_sharpe = float(np.max(full_sharpes))

    return PBOResult(
        pbo=pbo,
        logits=logits_arr,
        is_sharpes=is_arr,
        oos_sharpes=oos_arr,
        prob_oos_loss=prob_loss,
        degradation_slope=float(slope),
        degradation_intercept=float(intercept),
        historical_sharpe=best_overall_sharpe,
        mode="cscv",
    )

    def evaluate_benchmark(
    self,
    historical: BacktestResult | pd.Series | float,
    synthetic: MonteCarloResult | CPCVResult | pd.Series | np.ndarray,
) -> PBOResult:
    """
    Evaluates how a single strategy's historical Sharpe compares to a synthetic
    (`MonteCarloResult`) or combinatorial (`CPCVResult`) Sharpe distribution.
    """
    if isinstance(historical, BacktestResult):
        if "sharpe_ratio" in historical.tear_sheet.index:
            val = historical.tear_sheet.loc["sharpe_ratio", "Value"]
            hist_sr = float(val)  # type: ignore[arg-type]
        else:
            hist_sr = float(
                self._annualized_sharpes(
                    historical.returns.to_numpy()[:, None], self.risk_free_rate
                )[0]
            )
    elif isinstance(historical, pd.Series):
        hist_sr = float(
            self._annualized_sharpes(
                historical.dropna().to_numpy()[:, None], self.risk_free_rate
            )[0]
        )
    else:
        hist_sr = float(historical)

    if isinstance(synthetic, MonteCarloResult):
        if "sharpe_ratio" in synthetic.metrics_distribution.columns:
            sim_srs = synthetic.metrics_distribution["sharpe_ratio"].to_numpy(
                dtype=np.float64
            )
        else:
            sim_srs = self._annualized_sharpes(
                synthetic.returns_matrix.to_numpy(dtype=np.float64),
                self.risk_free_rate,
            )
    elif isinstance(synthetic, CPCVResult):
        sim_srs = synthetic.sharpe_ratios.to_numpy(dtype=np.float64)
    elif isinstance(synthetic, pd.Series):
        sim_srs = synthetic.to_numpy(dtype=np.float64)
    else:
        sim_srs = np.asarray(synthetic, dtype=np.float64)

    sim_srs = sim_srs[np.isfinite(sim_srs)]
    if len(sim_srs) == 0:
        sim_srs = np.array([0.0])

    # In single-strategy benchmark mode, overfitting risk is the probability mass of
    # synthetic/scrambled realizations falling below the historical Sharpe (or < 0),
    # and we also compute centered rank logits around the median synthetic Sharpe.
    pbo = float(np.mean(sim_srs < 0.0))
    prob_loss = float(np.mean(sim_srs < 0.0))

    ranks = (np.argsort(np.argsort(sim_srs)) + 0.5) / len(sim_srs)
    logits = np.log(ranks / (1.0 - ranks))
    is_srs = np.full_like(sim_srs, hist_sr)

    return PBOResult(
        pbo=pbo,
        logits=logits,
        is_sharpes=is_srs,
        oos_sharpes=sim_srs,
        prob_oos_loss=prob_loss,
        degradation_slope=0.0,
        degradation_intercept=float(np.mean(sim_srs)),
        historical_sharpe=hist_sr,
        mode="benchmark",
    )

    try:

                tracker._enabled = saved_enabled

            self._trials.append(rec)
        return rec

    def run(
    self,
    data: MarketData,
    scenario_data: ScenarioMarketData | None = None,
) -> ComparisonResult:
    """
    Executes all configured strategy pipelines concurrently over `data`, updates the
    `BacktestTracker` singleton, computes the rigorous Deflated Sharpe Ratio (DSR) across
    all trials, runs 3-pillar performance attribution, and returns a `ComparisonResult`.
    """
    tracker = BacktestTracker.instance()
    if self.reset_tracker:
        tracker.reset()

    names = list(self.pipelines.keys())
    engines = {name: self._build_engine(self.pipelines[name]) for name in names}

    def _exec_one(name: str) -> tuple[str, BacktestResult]:
        res = engines[name].run(data)
        res.name = name
        return name, res

    n_workers = self.max_workers or min(len(names), 8)
    raw_results: dict[str, BacktestResult] = {}
    if n_workers <= 1 or len(names) == 1:
        for name in names:
            _, r = _exec_one(name)
            raw_results[name] = r
    else:
        with ThreadPoolExecutor(max_workers=n_workers) as pool:
            futures = [pool.submit(_exec_one, name) for name in names]
            for fut in futures:
                k, r = fut.result()
                raw_results[k] = r

    # Preserve user dictionary order
    ordered_results: dict[str, BacktestResult] = {
        name: raw_results[name] for name in names
    }

    # Resolve benchmark returns & weights for cross-strategy comparison & attribution
    bench_result: BacktestResult | None = None
    bench_returns: pd.Series | None = None
    if isinstance(self.benchmark, str) and self.benchmark in ordered_results:
        bench_result = ordered_results[self.benchmark]
        bench_returns = bench_result.returns
    elif isinstance(self.benchmark, BacktestResult):
        bench_result = self.benchmark
        bench_returns = bench_result.returns
    elif isinstance(self.benchmark, pd.Series):
        bench_returns = self.benchmark
    else:
        # Default benchmark: Equal-Weight daily return across assets in `data`
        asset_rets = data.prices.pct_change().dropna()
        bench_returns = asset_rets.mean(axis=1)

    # Synchronize DSR across all strategies using post-batch tracker state (N & trial_sharpes)
    total_n_trials = max(tracker.n_trials, len(ordered_results))
    all_trial_sharpes = tracker.trial_sharpes
    if len(all_trial_sharpes) < len(ordered_results):
        all_trial_sharpes = [
            float(
                cast_val(r.tear_sheet.loc["sharpe_ratio", "Value"])
                if "sharpe_ratio" in r.tear_sheet.index
                else 0.0
            )
            for r in ordered_results.values()
        ]

    attributor = PerformanceAttributor()
    attributions: dict[str, AttributionResult] = {}
    summary_rows: dict[str, dict[str, float]] = {}

    for name, bt_res in ordered_results.items():
        updated_ts = MetricsTearSheet.generate(
            portfolio_returns=bt_res.returns,
            risk_free_rate=data.risk_free_rate,
            weights=bt_res.weights.reindex(bt_res.returns.index),

        )

            backtest_result=bt_res,
            market_data=data,
            benchmark_weights=bench_result,
            benchmark_returns=bench_returns,
            sector_map=self.sector_map,
            factor_returns=self.factor_returns,
        )
        attributions[name] = attr_res

        row_dict = {
            str(idx): float(cast_val(updated_ts.loc[idx, "Value"]))
            for idx in updated_ts.index
        }
        row_dict["n_trials"] = float(total_n_trials)
        row_dict["brinson_allocation"] = attr_res.brinson.total_allocation_effect
        row_dict["brinson_selection"] = attr_res.brinson.total_selection_effect
        row_dict["brinson_interaction"] = attr_res.brinson.total_interaction_effect
        row_dict["factor_alpha"] = attr_res.factor.annualized_alpha
        row_dict["factor_r_squared"] = attr_res.factor.r_squared
        row_dict["total_cost_drag"] = float(
            attr_res.frictional.waterfall_summary["total_explicit_cost_drag"]
        )
        row_dict["alpha_decay_drift_drag"] = float(
            attr_res.frictional.waterfall_summary["alpha_decay_and_drift_drag"]
        )
        summary_rows[name] = row_dict

    summary_df = pd.DataFrame.from_dict(summary_rows, orient="index")
    summary_df.index.name = "Strategy"

    scenario_results: dict[str, MonteCarloResult] | None = None
    if scenario_data is not None:
        rf_scalar = (
            float(data.risk_free_rate.mean())
            if isinstance(data.risk_free_rate, pd.Series)
            else float(data.risk_free_rate)
        )
        scenario_results = {
            name: self._evaluate_on_scenarios(
                bt_res, scenario_data, risk_free_rate=rf_scalar
            )
            for name, bt_res in ordered_results.items()
        }

    sr0_val = (
        float(summary_df["expected_max_sharpe_ratio"].iloc[0])
        if "expected_max_sharpe_ratio" in summary_df.columns and len(summary_df) > 0
        else 0.0
    )

    comp_res = ComparisonResult(
        summary_table=summary_df,
        results=ordered_results,
        attributions=attributions,
        scenario_results=scenario_results,
        n_trials=total_n_trials,
        expected_max_sharpe=sr0_val,
    )
    self.last_result = comp_res
    return comp_res

    def compute_deflated_sharpe_ratio(
    portfolio_returns: pd.Series | np.ndarray,
    n_trials: int = 1,
    trial_sharpes: Sequence[float] | np.ndarray | None = None,
    sharpe_variance: float | None = None,
    risk_free_rate: float | pd.Series = 0.0,
    target_sharpe: float = 0.0,
    ann_factor: float = 252.0,
) -> tuple[float, float]:
    """
    Computes the Deflated Sharpe Ratio (DSR) and annualized Expected Maximum Sharpe Ratio (SR_0)
    following Bailey & Lopez de Prado (2014), accounting for:
      1. Multiple testing penalty across `n_trials` (N) backtest variations,
      2. Track record length (T),
      3. Non-Normal return skewness (gamma_3) and Pearson kurtosis (gamma_4).

    Note: `trial_sharpes` and `sharpe_variance` (if provided) are expected in annualized Sharpe units.

    Returns:
        (dsr_probability, annualized_expected_max_sharpe)
    """
    if isinstance(portfolio_returns, pd.Series):
        clean = portfolio_returns.dropna()
    else:
        arr = np.asarray(portfolio_returns, dtype=np.float64)
        clean = pd.Series(arr[np.isfinite(arr)])

    n_obs = len(clean)
    if n_obs <= 2:
        return 0.0, float(target_sharpe)

    if isinstance(risk_free_rate, pd.Series):
        aligned_rf = risk_free_rate.reindex(clean.index).ffill().fillna(0.0)
        rf_annual = float(aligned_rf.mean()) if len(aligned_rf) > 0 else 0.0
    else:
        rf_annual = float(risk_free_rate)

    vals = clean.to_numpy(dtype=np.float64)
    cum_return = float(np.prod(1.0 + vals) - 1.0)
    n_years = max(n_obs / ann_factor, 1e-4)
    ann_ret = (
        float((1.0 + cum_return) ** (1.0 / n_years) - 1.0)
        if (1.0 + cum_return) > 0.0
        else -1.0
    )
    ann_vol = float(np.std(vals, ddof=1) * math.sqrt(ann_factor))
    if ann_vol <= 1e-12:
        return (1.0 if cum_return > 0.0 else 0.0), float(target_sharpe)

    sr_annual = (ann_ret - rf_annual) / ann_vol
    sr_daily = sr_annual / math.sqrt(ann_factor)

    skew_val = float(cast(Any, clean.skew())) if n_obs > 2 else 0.0
    kurt_val = float(cast(Any, clean.kurtosis())) if n_obs > 3 else 0.0
    skew_val = 0.0 if math.isnan(skew_val) else skew_val
    kurt_val = 0.0 if math.isnan(kurt_val) else kurt_val
    pearson_kurt = kurt_val + 3.0

    # Determine daily standard deviation of trial Sharpe ratios across N trials
    eff_n = max(int(n_trials), 1)
    if trial_sharpes is not None:
        ts_arr = np.asarray(list(trial_sharpes), dtype=np.float64)
        ts_arr = ts_arr[np.isfinite(ts_arr)]
        eff_n = max(eff_n, len(ts_arr))
        if len(ts_arr) >= 2:
            emp_std_ann = float(np.std(ts_arr, ddof=1))
            emp_std_daily = emp_std_ann / math.sqrt(ann_factor)
        else:
            emp_std_daily = 0.0
    elif sharpe_variance is not None and sharpe_variance > 0.0:
        emp_std_daily = math.sqrt(float(sharpe_variance)) / math.sqrt(ann_factor)
    else:
        emp_std_daily = 0.0

    # Under H_0, asymptotic standard error of daily Sharpe over T observations is 1 / sqrt(T - 1)
    null_se_daily = 1.0 / math.sqrt(max(n_obs - 1, 1))
    sr_std_daily = emp_std_daily if emp_std_daily > 1e-8 else null_se_daily

    base_sr0_daily = target_sharpe / math.sqrt(ann_factor)
    sr0_daily = compute_expected_max_sharpe(
        n_trials=eff_n,
        sharpe_std=sr_std_daily,
        mean_sharpe=base_sr0_daily,
    )
    sr0_annual = sr0_daily * math.sqrt(ann_factor)

    denom_sq = 1.0 - skew_val * sr_daily + ((pearson_kurt - 1.0) / 4.0) * (sr_daily**2)
    denom = math.sqrt(max(denom_sq, 1e-12))
    z_dsr = (sr_daily - sr0_daily) * math.sqrt(n_obs - 1) / denom
    dsr_val = float(norm.cdf(z_dsr))
    return dsr_val, float(sr0_annual)

    def compute_expected_max_sharpe(
    n_trials: int,
    sharpe_std: float,
    mean_sharpe: float = 0.0,
) -> float:
    """
    Computes the Expected Maximum Sharpe Ratio E[max_n SR_n] across `n_trials` independent
    backtest configurations under the False Strategy Theorem (Bailey & Lopez de Prado, 2014):

        E[max SR] = E[SR] + sqrt(Var[SR]) * ((1 - gamma) * Phi^{-1}(1 - 1/N) + gamma * Phi^{-1}(1 - 1/(N * e)))

    where gamma is the Euler-Mascheroni constant (~0.5772).
    """
    if n_trials <= 1 or sharpe_std <= 0.0:
        return float(mean_sharpe)

    n_float = float(max(n_trials, 2))
    q1 = float(norm.ppf(1.0 - 1.0 / n_float))
    q2 = float(norm.ppf(1.0 - 1.0 / (n_float * math.e)))
    z_max = (1.0 - EULER_MASCHERONI) * q1 + EULER_MASCHERONI * q2
    return float(mean_sharpe + sharpe_std * z_max)

    def resolve_sector_map(
    self,
    market_data: MarketData,
    assets: Sequence[str] | None = None,
    sector_map: Mapping[str, str] | pd.Series | pd.DataFrame | None = None,

    """
    Resolves asset-to-sector/cluster mappings using a 3-tier hierarchy:
      1. Explicit `sector_map` argument (dict, Series, or DataFrame).
      2. `market_data.cross_sectional` ('sector', 'group', 'cluster', 'industry', 'class').
      3. Automatic hierarchical correlation clustering (Ward / correlation distance)
         mirroring HRP / HERC / NCO cluster formation.
    """
    asset_list = (
        [str(a) for a in assets]
        if assets is not None
        else [str(a) for a in market_data.prices.columns]
    )

    resolved: dict[str, str] = {}

    # Tier 1: Explicit sector_map
    if sector_map is not None:
        if isinstance(sector_map, pd.DataFrame):
            col = sector_map.columns[0]
            for a in asset_list:
                if a in sector_map.index:
                    resolved[a] = str(sector_map.loc[a, col])
        elif isinstance(sector_map, pd.Series):
            for a in asset_list:
                if a in sector_map.index:
                    resolved[a] = str(sector_map.loc[a])
        else:
            for a in asset_list:
                if a in sector_map:
                    resolved[a] = str(sector_map[a])
        if resolved:
            for a in asset_list:
                resolved.setdefault(a, "Other")
            return resolved

    cs = market_data.cross_sectional
    if cs is not None and not cs.empty:
        candidate_cols = [
            c
            for c in cs.columns
            if any(
                kw in str(c).lower()
                for kw in ("sector", "group", "cluster", "industry", "class")
            )
        ]
        if candidate_cols:
            col = candidate_cols[0]
            for a in asset_list:
                if a in cs.index:
                    resolved[a] = str(cs.loc[a, col])
            if resolved:
                for a in asset_list:
                    resolved.setdefault(a, "Other")
                return resolved

    n_assets = len(asset_list)
    if n_assets <= 2:
        return {a: f"Cluster_{idx + 1}" for idx, a in enumerate(asset_list)}

    rets = market_data.prices.reindex(columns=asset_list).pct_change().dropna()
    if len(rets) > 3:
        corr = rets.corr().fillna(0.0).to_numpy(dtype=np.float64, copy=True)
        np.fill_diagonal(corr, 1.0)
        dist = np.sqrt(np.clip(0.5 * (1.0 - corr), 0.0, 1.0)).copy()
        np.fill_diagonal(dist, 0.0)
        try:
            condensed = ssd.squareform(dist, checks=False)
            link = sch.linkage(condensed, method="ward")
            k = min(self.n_auto_clusters, n_assets)
            labels = sch.fcluster(link, t=k, criterion="maxclust")
            return {
                asset_list[i]: f"Cluster_{int(labels[i])}" for i in range(n_assets)
            }
        except (ValueError, np.linalg.LinAlgError):
            pass

    k = min(self.n_auto_clusters, n_assets)
    return {asset_list[i]: f"Cluster_{(i % k) + 1}" for i in range(n_assets)}

    def build_proxy_factors(
    self,
    market_data: MarketData,
    factor_returns: pd.DataFrame | None = None,
) -> pd.DataFrame:
    """
    Returns aligned daily factor return series for Factor-Implied Attribution.
    Uses explicit `factor_returns` or `market_data.features['factors']` when provided;
    otherwise constructs canonical cross-sectional / time-series risk premia proxies
    directly from `market_data` (`Market`, `Momentum`, `LowVolatility`, plus `Size`
    and `Value` if available in `market_data.cross_sectional`).
    """
    asset_rets = market_data.prices.pct_change().dropna()
    if factor_returns is not None and not factor_returns.empty:
        return factor_returns.reindex(asset_rets.index).fillna(0.0)

    if (
        "factors" in market_data.features
        and not market_data.features["factors"].empty
    ):
        return market_data.features["factors"].reindex(asset_rets.index).fillna(0.0)

    n_dates, n_assets = asset_rets.shape
    if n_dates == 0:
        return pd.DataFrame(
            columns=["Market", "Momentum", "LowVolatility"], index=asset_rets.index
        )

    rf_daily: pd.Series | float
    if isinstance(market_data.risk_free_rate, pd.Series):
        rf_daily = (
            market_data.risk_free_rate.reindex(asset_rets.index).ffill().fillna(0.0)
            / self.ann_factor
        )
    else:
        rf_daily = float(market_data.risk_free_rate) / self.ann_factor

    factors_dict: dict[str, pd.Series] = {}

    # 1. Market Excess Return (MKT)
    mkt_ret = asset_rets.mean(axis=1) - rf_daily
    factors_dict["Market"] = mkt_ret

    if n_assets >= 2:
        # 2. Cross-Sectional Momentum (Winners minus Losers using lagged trailing returns)
        mom_win = min(60, max(5, n_dates // 4))
        lagged_cum_ret = (
            (1.0 + asset_rets)
            .rolling(window=mom_win, min_periods=2)
            .apply(np.prod, raw=True)
            .shift(1)
            .fillna(1.0)
        )
        ranks_mom = lagged_cum_ret.rank(axis=1, pct=True)
        demeaned_mom = ranks_mom.sub(ranks_mom.mean(axis=1), axis=0)
        denom_mom = demeaned_mom.abs().sum(axis=1).replace(0.0, 1.0)
        w_mom = demeaned_mom.div(denom_mom, axis=0) * 2.0
        factors_dict["Momentum"] = (w_mom * asset_rets).sum(axis=1)

        # 3. Cross-Sectional Low Volatility (BAB / Low-Vol minus High-Vol)
        vol_win = min(20, max(5, n_dates // 5))
        lagged_vol = (
            asset_rets.rolling(window=vol_win, min_periods=2)
            .std()
            .shift(1)
            .bfill()
            .fillna(0.01)
        )
        ranks_low_vol = (-lagged_vol).rank(axis=1, pct=True)
        demeaned_lv = ranks_low_vol.sub(ranks_low_vol.mean(axis=1), axis=0)
        denom_lv = demeaned_lv.abs().sum(axis=1).replace(0.0, 1.0)
        w_lv = demeaned_lv.div(denom_lv, axis=0) * 2.0
        factors_dict["LowVolatility"] = (w_lv * asset_rets).sum(axis=1)

        # 4. Optional Size (SMB) and Value (HML) from cross_sectional metadata
        cs = market_data.cross_sectional
        if cs is not None and not cs.empty:
            for col in cs.columns:
                col_lower = str(col).lower()
                if any(k in col_lower for k in ("market_cap", "mcap", "size")):
                    vals = pd.to_numeric(
                        cs[col].reindex(asset_rets.columns), errors="coerce"
                    ).dropna()
                    if len(vals) >= 2:
                        # Small minus Big
                        r_smb = (-vals).rank(pct=True)
                        w_smb = r_smb - float(r_smb.mean())
                        norm_s = float(w_smb.abs().sum())
                        if norm_s > 1e-12:
                            w_smb = (w_smb / norm_s) * 2.0
                            factors_dict["Size"] = (
                                asset_rets[vals.index]
                                .mul(w_smb, axis=1)
                                .sum(axis=1)
                            )
                elif any(
                    k in col_lower for k in ("book_to_market", "btm", "value", "pe")
                ):
                    vals = pd.to_numeric(
                        cs[col].reindex(asset_rets.columns), errors="coerce"
                    ).dropna()
                    if len(vals) >= 2:
                        sign = -1.0 if "pe" in col_lower else 1.0
                        r_val = (sign * vals).rank(pct=True)
                        w_val = r_val - float(r_val.mean())
                        norm_v = float(w_val.abs().sum())
                        if norm_v > 1e-12:
                            w_val = (w_val / norm_v) * 2.0
                            factors_dict["Value"] = (
                                asset_rets[vals.index]
                                .mul(w_val, axis=1)
                                .sum(axis=1)
                            )
    else:
        factors_dict["Momentum"] = pd.Series(0.0, index=asset_rets.index)
        factors_dict["LowVolatility"] = pd.Series(0.0, index=asset_rets.index)

    return pd.DataFrame(factors_dict, index=asset_rets.index).fillna(0.0)

    def compute_brinson_fachler(
    self,
    backtest_result: BacktestResult,
    market_data: MarketData,
    benchmark_weights: pd.DataFrame | pd.Series | BacktestResult | None = None,
    sector_map: Mapping[str, str] | pd.Series | pd.DataFrame | None = None,
    linking_method: str | None = None,
) -> BrinsonAttributionResult:
    """
    Computes Classical Brinson-Fachler (1985) active return attribution across clusters/sectors.
    Supports both Carino (1999) geometric multi-period linking and arithmetic linking.
    """
    method = linking_method or self.linking_method
    if method not in ("carino", "arithmetic"):
        raise ValueError("`linking_method` must be 'carino' or 'arithmetic'.")

    port_w_df = backtest_result.weights.dropna(how="all").fillna(0.0)
    asset_rets_all = market_data.prices.pct_change().dropna()
    common_dates = port_w_df.index.intersection(asset_rets_all.index)
    assets = [str(c) for c in port_w_df.columns if c in asset_rets_all.columns]

    resolved_map = self.resolve_sector_map(
        market_data, assets=assets, sector_map=sector_map
    )
    sectors = sorted(set(resolved_map.values()))

    if len(common_dates) == 0 or len(assets) == 0:
        empty_summary = pd.DataFrame(
            0.0,
            index=sectors,
            columns=[
                "portfolio_weight",
                "benchmark_weight",
                "active_weight",
                "portfolio_return",
                "benchmark_return",
                "allocation_effect",
                "selection_effect",
                "interaction_effect",
                "total_active_return",
            ],
        )
        empty_ts = pd.DataFrame(
            columns=[
                "allocation_effect",
                "selection_effect",
                "interaction_effect",
                "active_return",
                "portfolio_return",
                "benchmark_return",
            ]
        )
        return BrinsonAttributionResult(
            sector_summary=empty_summary,
            timeseries=empty_ts,
            total_allocation_effect=0.0,
            total_selection_effect=0.0,
            total_interaction_effect=0.0,
            total_active_return=0.0,
            linking_method=method,
            sector_map=resolved_map,
        )

    W_p = port_w_df.loc[common_dates, assets].to_numpy(dtype=np.float64)
    R_assets = asset_rets_all.loc[common_dates, assets].to_numpy(dtype=np.float64)
    T, N = W_p.shape

    # Normalize row sums of W_p if needed so Brinson identity holds cleanly
    row_sums_p = np.sum(W_p, axis=1, keepdims=True)
    W_p = np.where(np.abs(row_sums_p) > 1e-12, W_p / row_sums_p, 1.0 / N)

    # Resolve benchmark weights W_b of shape (T, N)
    if benchmark_weights is None:
        W_b = np.full((T, N), 1.0 / N, dtype=np.float64)
    elif isinstance(benchmark_weights, BacktestResult):
        bw_df = (
            benchmark_weights.weights.reindex(index=common_dates, columns=assets)
            .ffill()
            .bfill()
            .fillna(1.0 / N)
        )
        W_b = bw_df.to_numpy(dtype=np.float64)
    elif isinstance(benchmark_weights, pd.DataFrame):
        bw_df = (
            benchmark_weights.reindex(index=common_dates, columns=assets)
            .ffill()
            .bfill()
            .fillna(1.0 / N)
        )
        W_b = bw_df.to_numpy(dtype=np.float64)
    else:
        bw_ser = (
            pd.Series(benchmark_weights)
            .reindex(assets)
            .fillna(1.0 / N)
            .to_numpy(dtype=np.float64)
        )
        W_b = np.tile(bw_ser, (T, 1))

    row_sums_b = np.sum(W_b, axis=1, keepdims=True)
    W_b = np.where(np.abs(row_sums_b) > 1e-12, W_b / row_sums_b, 1.0 / N)

    n_sectors = len(sectors)
    # Binary membership matrix M of shape (N, n_sectors)
    M_sec = np.zeros((N, n_sectors), dtype=np.float64)
    for i, a in enumerate(assets):
        s_idx = sectors.index(resolved_map[a])
        M_sec[i, s_idx] = 1.0

    # Sector weights: (T, n_sectors)
    W_p_sec = W_p @ M_sec
    W_b_sec = W_b @ M_sec

    # Weighted asset returns within each sector: (T, n_sectors)
    num_p_sec = (W_p * R_assets) @ M_sec
    num_b_sec = (W_b * R_assets) @ M_sec

    # Sector return R_{p,j,t} and R_{b,j,t}
    # Note: When W_{p,j,t} == 0, set R_{p,j,t} = R_{b,j,t} so selection/interaction = 0
    R_b_sec = np.where(
        np.abs(W_b_sec) > 1e-12,
        num_b_sec / np.where(np.abs(W_b_sec) > 1e-12, W_b_sec, 1.0),
        0.0,
    )
    R_p_sec = np.where(
        np.abs(W_p_sec) > 1e-12,
        num_p_sec / np.where(np.abs(W_p_sec) > 1e-12, W_p_sec, 1.0),
        R_b_sec,
    )

    # Total portfolio and benchmark gross return per period t: (T,)
    R_p_t = np.sum(W_p_sec * R_p_sec, axis=1)
    R_b_t = np.sum(W_b_sec * R_b_sec, axis=1)

    # Classical Brinson-Fachler single-period effects: shape (T, n_sectors)
    A_t = (W_p_sec - W_b_sec) * (R_b_sec - R_b_t[:, None])
    S_t = W_b_sec * (R_p_sec - R_b_sec)
    I_t = (W_p_sec - W_b_sec) * (R_p_sec - R_b_sec)

    if method == "carino":
        # Carino (1999) logarithmic multi-period linking factors
        safe_rp = np.maximum(1.0 + R_p_t, 1e-8)
        safe_rb = np.maximum(1.0 + R_b_t, 1e-8)
        diff_t = R_p_t - R_b_t
        k_t = np.where(
            np.abs(diff_t) > 1e-10,
            (np.log(safe_rp) - np.log(safe_rb))
            / np.where(np.abs(diff_t) > 1e-10, diff_t, 1.0),
            1.0 / safe_rp,
        )

        cum_rp = float(np.prod(safe_rp) - 1.0)
        cum_rb = float(np.prod(safe_rb) - 1.0)
        total_active = cum_rp - cum_rb

        if abs(total_active) > 1e-10:
            K_total = (
                math.log(max(1.0 + cum_rp, 1e-8))
                - math.log(max(1.0 + cum_rb, 1e-8))
            ) / total_active
        else:
            K_total = 1.0 / max(1.0 + cum_rp, 1e-8)

        scale_t = k_t / max(K_total, 1e-12)
        # Enforce exact machine-precision summation if slight log-clipping occurred
        linked_sum = float(np.sum(scale_t * diff_t))
        if abs(linked_sum) > 1e-12 and abs(total_active) > 1e-12:
            scale_t = scale_t * (total_active / linked_sum)

        A_linked = A_t * scale_t[:, None]
        S_linked = S_t * scale_t[:, None]
        I_linked = I_t * scale_t[:, None]

        sec_p_ret = np.prod(1.0 + R_p_sec, axis=0) - 1.0
        sec_b_ret = np.prod(1.0 + R_b_sec, axis=0) - 1.0
    else:
        A_linked = A_t
        S_linked = S_t
        I_linked = I_t
        total_active = float(np.sum(R_p_t - R_b_t))
        sec_p_ret = np.sum(R_p_sec, axis=0)
        sec_b_ret = np.sum(R_b_sec, axis=0)

    sec_alloc = np.sum(A_linked, axis=0)
    sec_sel = np.sum(S_linked, axis=0)
    sec_inter = np.sum(I_linked, axis=0)
    sec_tot = sec_alloc + sec_sel + sec_inter

    avg_wp = np.mean(W_p_sec, axis=0)
    avg_wb = np.mean(W_b_sec, axis=0)

    sector_summary = pd.DataFrame(
        {
            "portfolio_weight": avg_wp,
            "benchmark_weight": avg_wb,
            "active_weight": avg_wp - avg_wb,
            "portfolio_return": sec_p_ret,
            "benchmark_return": sec_b_ret,
            "allocation_effect": sec_alloc,
            "selection_effect": sec_sel,
            "interaction_effect": sec_inter,
            "total_active_return": sec_tot,
        },
        index=sectors,
    )

    ts_alloc = np.sum(A_linked, axis=1)
    ts_sel = np.sum(S_linked, axis=1)
    ts_inter = np.sum(I_linked, axis=1)
    timeseries = pd.DataFrame(
        {
            "allocation_effect": ts_alloc,
            "selection_effect": ts_sel,
            "interaction_effect": ts_inter,
            "active_return": ts_alloc + ts_sel + ts_inter,
            "portfolio_return": R_p_t,
            "benchmark_return": R_b_t,
        },
        index=common_dates,
    )

    return BrinsonAttributionResult(
        sector_summary=sector_summary,
        timeseries=timeseries,
        total_allocation_effect=float(np.sum(sec_alloc)),
        total_selection_effect=float(np.sum(sec_sel)),
        total_interaction_effect=float(np.sum(sec_inter)),
        total_active_return=float(np.sum(sec_tot)),
        linking_method=method,
        sector_map=resolved_map,
    )

            )
        else:
            rf_arr = np.full(
                len(common_dates),
                float(market_data.risk_free_rate) / self.ann_factor,
            )
        y_target = y_port - rf_arr

    F_mat = factors_df.loc[common_dates, factor_cols].to_numpy(dtype=np.float64)
    T = len(common_dates)
    X_design = np.column_stack([np.ones(T, dtype=np.float64), F_mat])

    beta, se, t_stats, p_vals, r2, adj_r2 = self._ols_newey_west(
        y_target, X_design, max_lags=self.hac_lags
    )

    alpha_daily = float(beta[0])
    ann_alpha = alpha_daily * self.ann_factor
    factor_betas = beta[1:]

    ann_factor_premia = np.mean(F_mat, axis=0) * self.ann_factor
    ann_factor_contrib = factor_betas * ann_factor_premia

    all_exposures = np.concatenate([[alpha_daily], factor_betas])
    all_premia = np.concatenate([[self.ann_factor], ann_factor_premia])
    all_contrib = np.concatenate([[ann_alpha], ann_factor_contrib])

    total_ann_active = float(np.mean(y_target) * self.ann_factor)
    denom_active = total_ann_active if abs(total_ann_active) > 1e-10 else 1.0
    pct_active = all_contrib / denom_active

    idx_names = ["Alpha", *factor_cols]
    factor_summary = pd.DataFrame(
        {
            "exposure": all_exposures,
            "std_error": se,
            "t_stat": t_stats,
            "p_value": p_vals,
            "annualized_premia": all_premia,
            "annualized_contribution": all_contrib,
            "pct_of_active_return": pct_active,
        },
        index=idx_names,
    )

    # Daily cumulative factor return contributions
    daily_contrib_mat = np.column_stack(
        [
            np.full(T, alpha_daily),
            F_mat * factor_betas[None, :],
        ]
    )
    cum_contrib_df = pd.DataFrame(
        np.cumsum(daily_contrib_mat, axis=0),
        index=common_dates,
        columns=idx_names,
    )

    # Rolling factor exposures
    win = min(self.rolling_factor_window, max(10, T // 2))
    rolling_rows: list[np.ndarray] = []
    for t_idx in range(T):
        start_i = max(0, t_idx - win + 1)
        if t_idx - start_i + 1 < len(factor_cols) + 2:
            rolling_rows.append(factor_betas.copy())
        else:
            X_sub = X_design[start_i : t_idx + 1]
            y_sub = y_target[start_i : t_idx + 1]
            b_sub = np.linalg.pinv(X_sub) @ y_sub
            rolling_rows.append(b_sub[1:])

    rolling_exposures = pd.DataFrame(
        rolling_rows, index=common_dates, columns=factor_cols
    )

    return FactorAttributionResult(
        factor_summary=factor_summary,
        r_squared=r2,
        adj_r_squared=adj_r2,
        annualized_alpha=ann_alpha,
        rolling_exposures=rolling_exposures,
        cumulative_contributions=cum_contrib_df,
    )

    def compute_frictional_drag(
    self,
    backtest_result: BacktestResult,
    market_data: MarketData | None = None,
) -> FrictionalDragResult:
    """
    Decomposes the Gross-to-Net execution bridge (Ma & Smith, 2025):
      1. Theoretical Paper Return (un-drifted target weights)
      2. Alpha Decay & Drift / Threshold Delay Drag (R_executed_gross - R_paper)
      3. Gross Executed Return (R_executed_gross)
      4. Spread & Commission Drag (Linear L1 cost)
      5. Square-Root Market Impact Drag (Almgren-Chriss impact cost)
      6. Short Borrow Drag (Financing cost)
      7. Net Realized Return (R_net)
    """
    net_ret = backtest_result.returns.dropna()
    idx = net_ret.index

    if len(idx) == 0:
        empty_ser = pd.Series(
            0.0,
            index=[
                "theoretical_paper_return",
                "alpha_decay_and_drift_drag",
                "gross_executed_return",
                "spread_and_commission_drag",
                "market_impact_drag",
                "short_borrow_drag",
                "total_explicit_cost_drag",
                "net_realized_return",
                "annualized_paper_return",
                "annualized_gross_return",
                "annualized_net_return",
                "cost_retention_ratio",
            ],
        )
        return FrictionalDragResult(
            waterfall_summary=empty_ser,
            daily_drag_df=pd.DataFrame(),
            cost_retention_ratio=1.0,
        )

    costs_aligned = (
        backtest_result.costs.reindex(idx).fillna(0.0)
        if backtest_result.costs is not None
        else pd.Series(0.0, index=idx)
    )

    if backtest_result.gross_returns is not None:
        gross_ret = backtest_result.gross_returns.reindex(idx).fillna(
            net_ret + costs_aligned
        )
    else:
        gross_ret = net_ret + costs_aligned

    if backtest_result.paper_returns is not None:
        paper_ret = backtest_result.paper_returns.reindex(idx).fillna(gross_ret)
    elif (
        market_data is not None
        and backtest_result.target_weights is not None
        and not backtest_result.target_weights.empty
    ):
        asset_rets = market_data.prices.pct_change().reindex(idx).fillna(0.0)
        tw = (
            backtest_result.target_weights.reindex(
                index=idx, columns=asset_rets.columns
            )
            .ffill()
            .fillna(0.0)
        )
        paper_ret = (tw * asset_rets).sum(axis=1)
    else:
        paper_ret = gross_ret.copy()

    if (
        backtest_result.cost_breakdown is not None
        and not backtest_result.cost_breakdown.empty
    ):
        cb = backtest_result.cost_breakdown.reindex(idx).fillna(0.0)
        spread_s = cb.get("spread_cost", costs_aligned)
        impact_s = cb.get("impact_cost", pd.Series(0.0, index=idx))
        borrow_s = cb.get("borrow_cost", pd.Series(0.0, index=idx))
    else:
        spread_s = costs_aligned.copy()
        impact_s = pd.Series(0.0, index=idx)
        borrow_s = pd.Series(0.0, index=idx)

    drift_drag_daily = gross_ret - paper_ret
    daily_df = pd.DataFrame(
        {
            "paper_return": paper_ret,
            "gross_return": gross_ret,
            "net_return": net_ret,
            "alpha_decay_drift_drag": drift_drag_daily,
            "spread_cost": spread_s,
            "impact_cost": impact_s,
            "borrow_cost": borrow_s,
            "total_cost": costs_aligned,
        },
        index=idx,
    )

    # Arithmetic cumulative totals so the waterfall sums identically:
    # paper + drift_drag - spread - impact - borrow == net
    tot_paper = float(paper_ret.sum())
    tot_drift = float(drift_drag_daily.sum())
    tot_gross = float(gross_ret.sum())
    tot_spread = float(spread_s.sum())
    tot_impact = float(impact_s.sum())
    tot_borrow = float(borrow_s.sum())
    tot_cost = tot_spread + tot_impact + tot_borrow
    tot_net = float(net_ret.sum())

    n_years = max(len(idx) / self.ann_factor, 1e-4)
    cum_paper = float(np.prod(1.0 + paper_ret.to_numpy(dtype=np.float64)) - 1.0)
    cum_gross = float(np.prod(1.0 + gross_ret.to_numpy(dtype=np.float64)) - 1.0)
    cum_net = float(np.prod(1.0 + net_ret.to_numpy(dtype=np.float64)) - 1.0)

    ann_paper = float((1.0 + max(cum_paper, -0.9999)) ** (1.0 / n_years) - 1.0)
    ann_gross = float((1.0 + max(cum_gross, -0.9999)) ** (1.0 / n_years) - 1.0)
    ann_net = float((1.0 + max(cum_net, -0.9999)) ** (1.0 / n_years) - 1.0)

    retention = (
        float(tot_net / tot_gross)
        if abs(tot_gross) > 1e-8 and tot_gross > 0
        else (1.0 if tot_cost <= 1e-12 else 0.0)
    )

    summary_ser = pd.Series(
        {
            "theoretical_paper_return": tot_paper,
            "alpha_decay_and_drift_drag": tot_drift,
            "gross_executed_return": tot_gross,
            "spread_and_commission_drag": -tot_spread,
            "market_impact_drag": -tot_impact,
            "short_borrow_drag": -tot_borrow,
            "total_explicit_cost_drag": -tot_cost,
            "net_realized_return": tot_net,
            "annualized_paper_return": ann_paper,
            "annualized_gross_return": ann_gross,
            "annualized_net_return": ann_net,
            "cost_retention_ratio": retention,
        },
        name=backtest_result.name,
    )

    return FrictionalDragResult(
        waterfall_summary=summary_ser,
        daily_drag_df=daily_df,
        cost_retention_ratio=retention,
    )

        factor=factor_res,
        frictional=frictional_res,
    )
