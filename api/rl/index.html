    FinRL-Compatible Multi-Period Portfolio Optimization Environment (Research Section 7.1).

    State Space s_t

    Flattened vector containing: - Pre-trade drifted portfolio weights w_{t^-} (N,) - Lookback window of asset returns R_{t-L+1:t} (L * N,) - Optional stacked features F_{t} (K * N,) - Rolling volatility diagonal sigma_t (N,) - Mahalanobis turbulence index d_t and Bures-Wasserstein regime distance W_2 (2,)

    Continuous vector in [-1, 1]^N mapped onto the long-only simplex Delta^N (or temperature-scaled softmax), followed by optional thermodynamic regime gating and turbulence circuit-breaker.

    Differential log-return / risk-adjusted net portfolio return: r_t = log(1 + r_{p,t}^{net}) - (gamma / 2) * (r_{p,t}^{net})^2 - eta_w2 * W_2 * turnover_t

    @staticmethod
def project_action_to_simplex(action: np.ndarray) -> np.ndarray:
    """Map raw continuous action vector in R^N onto the probability simplex Delta^N."""
    arr = np.asarray(action, dtype=np.float64).flatten()
    if np.all(arr >= 0.0) and np.isclose(np.sum(arr), 1.0, atol=1e-4):
        return arr / np.sum(arr)
    shifted = arr - np.max(arr)
    exp_a = np.exp(np.clip(shifted, -20.0, 20.0))
    return exp_a / np.sum(exp_a)

    def get_current_tradability_mask(self) -> np.ndarray:
    """
    Returns the Point-in-Time active tradability mask ``(n_assets,)`` for the
    current lookback window ``[current_step - lookback_window, current_step)``.
    """
    start = max(0, self.current_step - self.lookback_window)
    end = max(start + 1, min(self.current_step, self.horizon))
    win_mask = self.tradability_mask[self.active_path_idx, start:end, :]

    if not np.any(step_mask):
        return np.ones(self.n_assets, dtype=bool)
    return step_mask

    def project_action(
    self,
    action: np.ndarray,
    deterministic: bool = True,
    mask: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Projects raw action vector in R^{action_dim} onto the simplex Delta^{action_dim},
    respecting the Point-in-Time tradability mask and returning `(risky_weights_N, cash_weight)`.
    """
    raw = np.asarray(action, dtype=np.float64).reshape(-1)
    if len(raw) < self.action_dim:
        pad = np.zeros(self.action_dim - len(raw), dtype=np.float64)
        raw = np.concatenate([raw, pad])
    elif len(raw) > self.action_dim:
        raw = raw[: self.action_dim]

    asset_mask = (
        np.asarray(mask, dtype=bool)
        if mask is not None
        else self.get_current_tradability_mask()
    )
    full_mask = (
        np.concatenate([asset_mask, np.array([True], dtype=bool)])
        if self.include_cash
        else asset_mask
    )
    m_f = full_mask.astype(np.float64)

    # Check if `raw` is already a valid masked simplex vector (e.g. from SetTransformerPolicyNet)
    if (
        np.all(raw >= -1e-7)
        and abs(float(np.sum(raw * m_f)) - 1.0) < 1e-4
        and float(np.sum(np.abs(raw * (1.0 - m_f)))) < 1e-5
    ):
        simplex_w = np.maximum(raw, 0.0) * m_f
        simplex_w = simplex_w / max(float(np.sum(simplex_w)), 1e-12)
    elif self.projection_mode == "relu_l1":
        clipped = np.maximum(raw, 0.0) * m_f
        total = float(np.sum(clipped))
        if total > 1e-12:
            simplex_w = clipped / total
        else:
            simplex_w = m_f / max(float(np.sum(m_f)), 1.0)
    elif self.projection_mode == "dirichlet":
        # Masked Dirichlet concentration alpha_i = (log(1 + exp(a_i)) + 1.0) * m_i
        clipped_raw = np.clip(raw, -20.0, 20.0)
        alpha = (np.log1p(np.exp(clipped_raw)) + 1.0) * m_f
        if deterministic:
            simplex_w = alpha / max(float(np.sum(alpha)), 1e-12)
        else:
            simplex_w = np.zeros(self.action_dim, dtype=np.float64)
            active_idx = np.flatnonzero(full_mask)
            if len(active_idx) > 0:
                simplex_w[active_idx] = np.random.dirichlet(alpha[active_idx])
    else:
        # Default: mask-aware temperature-scaled softmax
        active_raw = np.where(full_mask, raw, -1e9)
        scaled = (active_raw - np.max(active_raw)) / self.softmax_temperature
        exp_w = np.exp(np.clip(scaled, -25.0, 25.0)) * m_f
        simplex_w = exp_w / max(float(np.sum(exp_w)), 1e-12)

    if self.include_cash:
        risky_w = simplex_w[: self.n_assets]
        cash_w = float(simplex_w[self.n_assets])
    else:
        risky_w = simplex_w[: self.n_assets]
        r_sum = float(np.sum(risky_w))
        if r_sum > 0:
            risky_w = risky_w / r_sum
        cash_w = 0.0

    return risky_w, cash_w

    def filter_series(self, returns: pd.DataFrame) -> pd.DataFrame:
    """
    Runs causal continuous-time Wonham SDE filtering across `returns` and returns
    a `(T, K)` DataFrame of posterior regime probabilities `p_t in Delta^K`.
    """
    if not self.is_fitted:
        self.fit(returns)
    assert (
        self.regime_means_ is not None
        and self.regime_covs_ is not None
        and self.generator_matrix_ is not None
    )

    clean_ret = returns.dropna(how="all").fillna(0.0)
    ret_np = clean_ret.to_numpy(dtype=np.float64)
    t_obs, n_assets = ret_np.shape
    k_reg = len(self.regime_means_)
    dt = 1.0 / self.annualization_factor

    p_curr = np.full(k_reg, 1.0 / k_reg, dtype=np.float64)
    beliefs = np.empty((t_obs, k_reg), dtype=np.float64)

    prec_mats = [
        np.linalg.inv(self.regime_covs_[k] + 1e-5 * np.eye(n_assets))
        for k in range(k_reg)
    ]

    for t in range(t_obs):
        r_t = ret_np[t]
        mu_bar = p_curr @ self.regime_means_
        cov_bar = sum(
            p_curr[k] * self.regime_covs_[k] for k in range(k_reg)
        ) + 1e-5 * np.eye(n_assets)
        prec_bar = np.linalg.inv(cov_bar)
        innov = r_t - mu_bar

        # Wonham SDE drift + innovation martingale update
        prior_drift = (p_curr @ self.generator_matrix_) * dt
        innov_gain = np.array(
            [
                p_curr[k]
                * float((self.regime_means_[k] - mu_bar) @ prec_bar @ innov)
                for k in range(k_reg)
            ],
            dtype=np.float64,
        )
        # Combine with quadratic log-likelihood innovation for second-order volatility discrimination
        quad_ll = np.array(
            [
                -0.5
                * float(
                    (r_t - self.regime_means_[k])
                    @ prec_mats[k]
                    @ (r_t - self.regime_means_[k])
                )
                - 0.5 * float(np.linalg.slogdet(self.regime_covs_[k])[1])
                for k in range(k_reg)
            ],
            dtype=np.float64,
        )
        quad_ll = quad_ll - np.max(quad_ll)
        like_weights = np.exp(np.clip(quad_ll, -20.0, 0.0))

        p_euler = np.maximum(p_curr + prior_drift + innov_gain, 1e-6)
        p_euler = p_euler / np.sum(p_euler)
        p_post = 0.65 * p_euler + 0.35 * (p_euler * like_weights)
        p_curr = np.clip(p_post / np.sum(p_post), 1e-4, 1.0)
        p_curr = p_curr / np.sum(p_curr)
        beliefs[t] = p_curr

    self.last_belief_ = p_curr.copy()
    cols = [f"regime_{k}" for k in range(k_reg)]
    return pd.DataFrame(beliefs, index=clean_ret.index, columns=cols)

    mu2: np.ndarray,
    cov2: np.ndarray,

    """
    Compute exact Bures-Wasserstein W_2 distance between N(mu1, cov1) and N(mu2, cov2).
    """
    n = len(mu1)
    eps_eye = np.eye(n, dtype=float) * 1e-8
    c1 = 0.5 * (cov1 + cov1.T) + eps_eye
    c2 = 0.5 * (cov2 + cov2.T) + eps_eye

    mean_diff_sq = float(np.sum((mu1 - mu2) ** 2))

    sqrt_c1 = np.real(sla.sqrtm(c1))
    inner = sqrt_c1 @ c2 @ sqrt_c1
    inner = 0.5 * (inner + inner.T) + eps_eye
    sqrt_inner = np.real(sla.sqrtm(inner))

    bures_sq = float(np.trace(c1 + c2 - 2.0 * sqrt_inner))
    w2_sq = max(0.0, mean_diff_sq + bures_sq)
    return float(np.sqrt(w2_sq))

        w_gated = damping * w_target + (1 - damping) * w_current
    """
    diag = self.evaluate(returns)
    if current_weights is None:
        return target_weights, diag

    w_curr = current_weights.reindex(target_weights.index).fillna(0.0)
    c_sum = float(w_curr.sum())
    if c_sum <= 0:
        return target_weights, diag
    w_curr = w_curr / c_sum

    lam = 0.0 if diag.should_halt_rebalance else diag.damping_factor
    gated = lam * target_weights + (1.0 - lam) * w_curr
    g_sum = float(gated.sum())
    if g_sum > 0:
        gated = gated / g_sum
    return gated, diag

    @staticmethod
def compute_subpath_log_signatures(
    returns_arr: np.ndarray,
    subpath_window: int = 8,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Extracts sliding-window depth-2 Lyndon log-signatures from a 2D return array (T, N).

    Returns:
        full_logsigs: shape (M, d + d(d-1)/2) where d = N + 1 (time-augmented)
        levy_areas: shape (M, d(d-1)/2)
    """
    t_len, n_assets = returns_arr.shape
    w = min(max(2, subpath_window), t_len)
    if t_len < w:
        d = n_assets + 1
        n_pairs = (d * (d - 1)) // 2
        return np.zeros((1, d + n_pairs), dtype=np.float64), np.zeros(
            (1, n_pairs), dtype=np.float64
        )

    vol_scale = np.maximum(np.std(returns_arr, axis=0, keepdims=True), 1e-6)
    norm_ret = np.clip(returns_arr / vol_scale, -10.0, 10.0)
    d = n_assets + 1
    pair_i, pair_j = np.triu_indices(d, k=1)
    n_sub = t_len - w + 1
    full_list: list[np.ndarray] = []
    levy_list: list[np.ndarray] = []
    dt_col = np.full((w, 1), 1.0 / float(w), dtype=np.float64)

    for start in range(n_sub):
        seg = norm_ret[start : start + w]
        dx = np.hstack([dt_col, seg])  # (w, d)
        x_cum = np.cumsum(dx, axis=0)
        x_prev = np.vstack([np.zeros((1, d), dtype=np.float64), x_cum[:-1]])
        s1 = x_cum[-1]
        levy_mat = 0.5 * (
            np.einsum("ti,tj->ij", x_prev, dx) - np.einsum("tj,ti->ij", x_prev, dx)
        )
        levy_vec = levy_mat[pair_i, pair_j]
        full_list.append(np.concatenate([s1, levy_vec]))
        levy_list.append(levy_vec)

    return np.vstack(full_list), np.vstack(levy_list)

    Parameters
    ----------
    return_window : torch.Tensor
        Recent asset return window of shape (B, L, N, F), (B, L, N), or (B, L * N).
    w_current : torch.Tensor, optional

        Target portfolio weights w_t on the simplex of shape (B, N).

    def compute_trajectory_loss(
    self,
    scenarios: torch.Tensor,
    w_init: torch.Tensor | None = None,
) -> torch.Tensor:
    """
    Unroll policy along a batch of return trajectories of shape (B, L + H, N)
    and compute the pathwise entropic + Sharpe trajectory loss.
    """
    B, T_total, N = scenarios.shape
    L = self.lookback_window
    H = T_total - L
    if H <= 0:
        raise ValueError(
            f"Trajectory length {T_total} must exceed lookback_window {L}."
        )

    if w_init is None:
        w_curr = torch.full(
            (B, N), 1.0 / N, device=scenarios.device, dtype=scenarios.dtype
        )
    else:
        w_curr = w_init

    net_returns_list: list[torch.Tensor] = []

    for s in range(H):
        ret_win = scenarios[:, s : s + L, :]  # (B, L, N)
        r_next = scenarios[:, s + L, :]  # (B, N)

        w_target = self.policy(ret_win, w_curr)
        turnover = torch.abs(w_target - w_curr).sum(dim=-1)  # (B,)
        tc = self.tc_rate * turnover

        gross_ret = (w_target * r_next).sum(dim=-1)  # (B,)
        net_ret = (1.0 - tc) * (1.0 + gross_ret) - 1.0
        net_returns_list.append(net_ret)

        # Differentiable post-return weight drift w_{(s+1)^-}
        grown = w_target * (1.0 + r_next).clamp(min=1e-4)
        w_curr = grown / grown.sum(dim=-1, keepdim=True)

    path_returns = torch.stack(net_returns_list, dim=1)  # (B, H)
    cum_return = path_returns.sum(dim=1)  # (B,)

    # Entropic risk measure (certainty equivalent) + path variance penalty
    gamma = self.risk_aversion
    scaled = torch.clamp(-gamma * cum_return, min=-20.0, max=20.0)
    entropic_loss = (torch.logsumexp(scaled, dim=0) - float(np.log(B))) / max(
        gamma, 1e-6
    )
    var_penalty = 0.5 * gamma * path_returns.var(dim=1, unbiased=False).mean()
    return entropic_loss + var_penalty

    def fit_on_market_data(
    self,
    data: MarketData,
    epochs: int = 20,
    batch_size: int = 16,
) -> list[float]:
    """
    Slice rolling trajectories of length (L + H) from `MarketData` and train `self.policy`.
    """
    returns_np = data.prices.pct_change().fillna(0.0).to_numpy(dtype=np.float32)
    L = self.lookback_window
    seq_len = L + self.horizon
    n_obs = len(returns_np)
    if n_obs <= seq_len:
        raise ValueError(
            f"Need more than {seq_len} observations in MarketData, got {n_obs}."
        )

    windows = np.stack(
        [returns_np[i : i + seq_len] for i in range(n_obs - seq_len + 1)],
        axis=0,
    )
    dataset_tensor = torch.tensor(windows, dtype=torch.float32, device=self.device)

    self.policy.train()
    loss_history: list[float] = []
    n_samples = dataset_tensor.shape[0]

    for _ in range(epochs):
        perm = torch.randperm(n_samples, device=self.device)
        epoch_losses: list[float] = []
        for start in range(0, n_samples, batch_size):
            batch_idx = perm[start : start + batch_size]
            batch_paths = dataset_tensor[batch_idx]

            self.optimizer.zero_grad()
            loss = self.compute_trajectory_loss(batch_paths)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(self.policy.parameters(), max_norm=1.0)
            self.optimizer.step()
            epoch_losses.append(float(loss.item()))

        loss_history.append(float(np.mean(epoch_losses)))

    self.policy.eval()
    return loss_history
